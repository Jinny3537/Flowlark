# Flowlark v0.3 Feedback and Import Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Deliver contextual prototype feedback, Markdown fallback, GitHub/GitLab/Gitee Issue creation, four HTML import paths, and a reliable watch inbox.

**Architecture:** Normalize feedback before sending it to provider adapters. Keep unsent feedback in local cache, keep provider secrets outside Git, and route all repository mutations through `Hub`.

**Tech Stack:** Node.js built-in `fetch`, `dns`, `crypto`, `child_process`, Vue 3, Ant Design Vue, Node test runner, local mock HTTP servers.

```js
import fs from 'node:fs'
import path from 'node:path'
import crypto from 'node:crypto'
import { INTERNAL_DIR } from './repo.js'
import { err } from './errors.js'
import { parse, stringify } from './json.js'

function validAnchor(anchor) {
  const keys = ['x', 'y', 'width', 'height']
  const out = Object.fromEntries(keys.map((key) => [key, Number(anchor && anchor[key])]))
  if (keys.some((key) => !Number.isFinite(out[key]) || out[key] < 0 || out[key] > 1)) {
    throw err.bad('FEEDBACK_ANCHOR_INVALID', '标注区域必须使用 0 到 1 的相对坐标')
  }
  if (out.width === 0 || out.height === 0 || out.x + out.width > 1 || out.y + out.height > 1) {
    throw err.bad('FEEDBACK_ANCHOR_INVALID', '标注区域超出原型范围')
  }
  return out
}

export function normalizeFeedback(input) {
  for (const key of ['title', 'description', 'project', 'version']) {
    if (!String(input[key] || '').trim()) throw err.bad('FEEDBACK_INVALID', `反馈缺少 ${key}`)
  }
  return {
    id: input.id || crypto.randomUUID(),
    title: String(input.title).trim(),
    description: String(input.description).trim(),
    project: String(input.project).trim(),
    version: String(input.version).trim(),
    baseline: input.baseline ? String(input.baseline) : null,
    requirements: [...new Set((input.requirements || []).map(String).filter(Boolean))],
    changes: Array.isArray(input.changes) ? input.changes : [],
    url: String(input.url || ''),
    anchor: validAnchor(input.anchor),
    createdAt: input.createdAt || new Date().toISOString()
  }
}

export function encodeAnchor(anchor) {
  return Buffer.from(JSON.stringify(validAnchor(anchor))).toString('base64url')
}

export function decodeAnchor(value) {
  try { return validAnchor(JSON.parse(Buffer.from(value, 'base64url').toString('utf8'))) }
  catch { throw err.bad('FEEDBACK_ANCHOR_INVALID', '标注链接无效') }
}

function cell(value) { return String(value == null ? '' : value).replaceAll('|', '\\|').replaceAll('\n', ' ') }

export function renderFeedbackMarkdown(item) {
  const rows = [
    ['项目', item.project], ['版本', item.version], ['当前基线', item.baseline || '无'],
    ['需求', item.requirements.join(', ') || '无'], ['原型链接', item.url || '无'],
    ['标注区域', JSON.stringify(item.anchor)]
  ]
  return `# ${item.title}\n\n${item.description}\n\n| 上下文 | 值 |\n|---|---|\n${rows.map(([k, v]) => `| ${cell(k)} | ${cell(v)} |`).join('\n')}\n`
}

function draftDir(root) { return path.join(root, INTERNAL_DIR, 'cache', 'feedback') }

export function saveDraft(root, input, screenshot = null) {
  const item = normalizeFeedback(input)
  const dir = draftDir(root)
  fs.mkdirSync(dir, { recursive: true })
  fs.writeFileSync(path.join(dir, `${item.id}.json`), stringify(item))
  if (screenshot) fs.writeFileSync(path.join(dir, `${item.id}.png`), screenshot)
  return item
}

export function listDrafts(root) {
  const dir = draftDir(root)
  if (!fs.existsSync(dir)) return []
  return fs.readdirSync(dir).filter((name) => name.endsWith('.json'))
    .map((name) => parse(fs.readFileSync(path.join(dir, name), 'utf8')))
    .sort((a, b) => b.createdAt.localeCompare(a.createdAt))
}

export function removeDraft(root, id) {
  for (const ext of ['.json', '.png']) fs.rmSync(path.join(draftDir(root), `${id}${ext}`), { force: true })
}
```

```js
import { execFileSync } from 'node:child_process'

export function getSecret(service, account, envKey) {
  if (envKey && process.env[envKey]) return process.env[envKey]
  if (process.platform !== 'darwin') return null
  try {
    return execFileSync('security', ['find-generic-password', '-s', service, '-a', account, '-w'], {
      encoding: 'utf8', stdio: ['ignore', 'pipe', 'pipe']
    }).trim() || null
  } catch (e) {
    if (e.status === 44) return null
    throw e
  }
}

export function setSecret(service, account, value) {
  if (process.platform !== 'darwin') throw new Error('当前系统不支持 macOS 钥匙串')
  execFileSync('security', ['add-generic-password', '-U', '-s', service, '-a', account, '-w', value], {
    stdio: ['ignore', 'ignore', 'pipe']
  })
}

export function deleteSecret(service, account) {
  if (process.platform !== 'darwin') return false
  try {
    execFileSync('security', ['delete-generic-password', '-s', service, '-a', account], {
      stdio: ['ignore', 'ignore', 'pipe']
    })
    return true
  } catch (e) {
    if (e.status === 44) return false
    throw e
  }
}
```
